  export async function joinKumaranFamily(customerId: string): 
  Promise<boolean> {
    const mutation = `
      mutation customerUpdate($input: CustomerInput!) {
        customerUpdate(input: $input) {
          customer {
            id
            tags
            emailMarketingConsent {
              marketingState
            }
          }
          userErrors {
            field
            message
          }
        }
      }
    `;

    // First get current customer to preserve existing tags
    const customer = await getCustomerById(customerId);
    if (!customer) return false;

    // Add kumaran-family tag if not already present
    const existingTags = customer.tags || [];
    const newTags = existingTags.includes("kumaran-family")
      ? existingTags
      : [...existingTags, "kumaran-family"];

    try {
      const data = await adminApiRequest<{
        customerUpdate: {
          customer: { id: string; tags: string[]; emailMarketingConsent: {
  marketingState: string } | null } | null;
          userErrors: Array<{ field: string; message: string }>;
        };
      }>(mutation, {

        }

      if (data.customerUpdate.userErrors.length > 0) {
